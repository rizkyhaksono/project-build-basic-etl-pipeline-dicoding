# ============================================================
# Cara Menjalankan Proyek ETL Pipeline - Fashion Studio
# ============================================================
# Proyek ini menggunakan uv (https://docs.astral.sh/uv/) sebagai
# package manager. Pastikan uv sudah terpasang:
#   curl -LsSf https://astral.sh/uv/install.sh | sh

# 1. Buat virtual environment dengan uv
uv venv --python 3.12

# 2. Aktifkan virtual environment
source .venv/bin/activate        # Linux/Mac
# .venv\Scripts\activate         # Windows

# 3. Install dependencies dari requirements.txt
uv pip install -r requirements.txt

# ------------------------------------------------------------
# Menjalankan skrip ETL pipeline
# ------------------------------------------------------------
python3 main.py
# atau tanpa mengaktifkan venv:
# uv run main.py

# ------------------------------------------------------------
# Menjalankan unit test pada folder tests
# ------------------------------------------------------------
python3 -m pytest tests
# atau: uv run pytest tests

# ------------------------------------------------------------
# Menjalankan test coverage pada folder tests
# ------------------------------------------------------------
coverage run -m pytest tests
coverage report -m
# atau (pytest-cov): uv run pytest tests --cov=utils
